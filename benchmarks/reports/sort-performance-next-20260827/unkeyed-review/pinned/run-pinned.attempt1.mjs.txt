function inventory(root, prefix = '') {
  return readdirSync(join(root, prefix), { withFileTypes: true }).flatMap(entry => {

    return entry.isDirectory() ? inventory(root, path) : [{ path, sha256: hash(readFileSync(join(root, path))) }];

}
