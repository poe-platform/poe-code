import { createGitCommand, createGitCommands, gitCommands } from '__MODULE_SPECIFIER__';
import type { CommandDefinition } from '__COMMAND_CONTRACT__';
import type { VirtualShellPlugin } from '__PLUGIN_CONTRACT__';

const command: CommandDefinition = createGitCommand({ discoveryBoundary: '/repo', replace: false });
const commands: readonly CommandDefinition[] = createGitCommands();
const plugin: VirtualShellPlugin = gitCommands({ replace: true });
const handler: CommandDefinition['execute'] = command.execute;
void [commands, plugin, handler];

// @ts-expect-error Ratified M1 API has no public limits override.
createGitCommand({ limits: { maxObjects: 1 } });
// @ts-expect-error A native process fallback is not an admitted API.
gitCommands({ spawn: () => undefined });
// @ts-expect-error Boundary must be a string.
createGitCommands({ discoveryBoundary: 1 });
