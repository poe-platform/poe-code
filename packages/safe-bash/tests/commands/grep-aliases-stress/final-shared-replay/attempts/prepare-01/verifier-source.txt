function command(name, executable, args, cwd = destination, timeout = 120000, allowFailure = false, input) {
  const child = spawnSync(executable, args, { cwd, env: environment, timeout, maxBuffer: 16 * 1024 * 1024, killSignal: 'SIGKILL', input });
  const captured = { name, executable, args, cwd, pid: child.pid, status: child.status, signal: child.signal, error: child.error?.message ?? null, stdoutHex: (child.stdout ?? Buffer.alloc(0)).toString('hex'), stderrHex: (child.stderr ?? Buffer.alloc(0)).toString('hex') };
  receipt.commands.push(captured);
  save(`${name}.json`, captured);
  assert.equal(child.error, undefined, name);
  assert.equal(child.signal, null, name);
  if (!allowFailure) assert.equal(child.status, 0, `${name}: ${child.stderr?.toString()} ${child.stdout?.toString()}`);
  return child;
}
function inventory(directory, relative = '') {
  return readdirSync(join(directory, relative)).sort().flatMap(name => {
    const path = join(relative, name);
    const absolute = join(directory, path);
    const metadata = lstatSync(absolute);
    if (metadata.isDirectory()) return [{ path, kind: 'directory' }, ...inventory(directory, path)];
    if (metadata.isSymbolicLink()) {
      const target = readlinkSync(absolute);
      return [{ path, kind: 'symlink', target, size: Buffer.byteLength(target), sha256: sha256(target) }];
    }
    assert.ok(metadata.isFile(), path);
    return [{ path, kind: 'file', mode: metadata.mode & 0o777, size: metadata.size, sha256: sha256(readFileSync(absolute)) }];
  });
}
function digestInventory(entries) {
  return { entries: entries.length, files: entries.filter(entry => entry.kind === 'file').length, symlinks: entries.filter(entry => entry.kind === 'symlink').length, sha256: sha256(JSON.stringify(entries)) };
}
function verifySeals(root) {
  return Object.entries(seals).map(([path, digest]) => {
    const file = join(root, path);
    const bytes = readFileSync(file);
    assert.equal(sha256(bytes), digest, path);
    const rows = bytes.toString().trim().split('\n');
    for (const row of rows) {
      const match = /^([a-f0-9]{64})  (.+)$/.exec(row);
      assert.ok(match);
      assert.equal(sha256(readFileSync(join(dirname(file), match[2]))), match[1], match[2]);
    }
    return { path, sha256: digest, payloads: rows.length };
  });
}
function verifyGit(entries, rows) {
  assert.deepEqual(entries.filter(entry => entry.kind !== 'directory').map(entry => entry.path).sort(), rows.map(row => row.path).sort());
  for (const row of rows) {
    const absolute = join(source, row.path);
    const metadata = lstatSync(absolute);
    const bytes = row.mode === '120000' ? Buffer.from(readlinkSync(absolute)) : readFileSync(absolute);
    assert.equal(metadata.isSymbolicLink(), row.mode === '120000', row.path);
    if (row.path.startsWith('src/')) assert.ok(metadata.isFile() && !metadata.isSymbolicLink(), row.path);
    assert.equal(createHash('sha1').update(`blob ${bytes.length}\0`).update(bytes).digest('hex'), row.blob, row.path);
  }
}
