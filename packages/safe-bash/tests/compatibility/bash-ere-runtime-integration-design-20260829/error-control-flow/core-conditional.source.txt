163:export async function evaluateConditional(expression: ConditionalExpression, context: ConditionalContext): Promise<boolean> {
164:  const stack: { node: ConditionalExpression; stage: number }[] = [{ node: expression, stage: 0 }];
165:  let result = false;
166:  while (stack.length) {
167:    await charge(context);
168:    const frame = stack.at(-1)!, node = frame.node;
169:    if (node.kind === "not") {
170:      if (frame.stage++ === 0) stack.push({ node: node.operand, stage: 0 });
171:      else { result = !result; stack.pop(); }
172:    } else if (node.kind === "and" || node.kind === "or") {
173:      if (frame.stage === 0) { frame.stage = 1; stack.push({ node: node.left, stage: 0 }); }
174:      else if (frame.stage === 1 && (node.kind === "and" ? result : !result)) { frame.stage = 2; stack.push({ node: node.right, stage: 0 }); }
175:      else stack.pop();
176:    } else { result = await leaf(node, context); stack.pop(); }
177:  }
178:  context.signal.throwIfAborted();
179:  return result;
