21:class RootInvocationCancellationOwner {
22:  readonly finalized: Promise<void>;
23:  #resolveFinalized!: () => void;
24:  #admissionOpen = true;
25:  #boundary: CancellationBoundary | undefined;
26:  #observedOrigin: CancellationOrigin | undefined;
27:  #captureCancellation: ((origin: CancellationOrigin) => void) | undefined;
28:  #detach: (() => void) | undefined;
29:  #finished = false;
30:
31:  constructor(readonly scope: InvocationScope) {
32:    this.finalized = new Promise<void>(resolve => { this.#resolveFinalized = resolve; });
33:    scope.register(() => { this.#admissionOpen = false; });
34:  }
35:
36:  activate(boundary: CancellationBoundary): void {
37:    if (!this.#admissionOpen) throw new Error("Root cancellation admission is closed");
38:    this.#boundary = boundary;
39:    this.#detach = subscribeCancellation(boundary, origin => { this.#captureCancellation?.(origin); });
40:  }
41:
42:  assertAdmissionOpen(): void {
43:    if (!this.#admissionOpen) throw new Error("Root cancellation admission is closed");
44:  }
45:
46:  capture<Value>(execute: () => Promise<Value>): Promise<CapturedCancellationOutcome<Value>> {
47:    return new Promise(resolve => {
48:      let settled = false;
49:      let raw: Promise<Value> | undefined;
50:      let queuedOrigin = false;
51:      const settle = (captured: CapturedCancellationOutcome<Value>): void => {
52:        if (settled) return;
53:        settled = true;
54:        this.#captureCancellation = undefined;
55:        resolve(captured);
56:      };
57:      this.#captureCancellation = origin => {
58:        if (settled || queuedOrigin) return;
59:        queuedOrigin = true;
60:        queueMicrotask(() => {
61:          if (settled) return;
62:          this.#observedOrigin = origin;
63:          settle({ kind: "throw", reason: origin.signal.reason });
64:          void raw?.catch(() => undefined);
65:        });
66:      };
67:      try { raw = Promise.resolve(execute()); }
70:        value => settle({ kind: "return", value }),
71:        reason => settle({ kind: "throw", reason }),
72:      );
73:      if (settled) void raw.catch(() => undefined);
74:    });
75:  }
76:
77:  finish<Value>(captured: CapturedCancellationOutcome<Value>): CancellationSelection<Value> {
78:    if (this.#finished) throw new Error("Root cancellation was already finalized");
79:    this.#finished = true;
80:    this.#admissionOpen = false;
81:    try {
82:      try { this.#detach?.(); } catch (error) { this.scope.failures.push(error); }
83:      this.#detach = undefined;
84:      const close = this.#boundary!.close();
85:      this.scope.failures.push(...close.failures);
86:      return selectRuntimeCancellationOutcome(this.#boundary!, captured, this.#observedOrigin);
87:    } finally { this.#resolveFinalized(); }
88:  }
89:}
