import assert from "node:assert/strict";
import { spawnSync } from "node:child_process";
import { createHash } from "node:crypto";
import { existsSync, mkdirSync, mkdtempSync, readFileSync, rmSync, symlinkSync, writeFileSync } from "node:fs";
import { dirname, join, resolve } from "node:path";
import { fileURLToPath } from "node:url";

const own = dirname(fileURLToPath(import.meta.url));
const root = resolve(own, "../../../..");
const revision = "3bf672f722da2bdf1591ed112290b702987bf63a";
const serviceRevision = "b22d00c2834358dc3083de58774f3aa188093f9b";
const evidence = join(own, "evidence", process.argv[2] ?? "");
assert.match(process.argv[2] ?? "", /^[a-z][a-z0-9-]*$/);
assert.equal(existsSync(evidence), false);
mkdirSync(evidence, { recursive: true });
const scratch = mkdtempSync(join(own, ".audit-"));
const hash = bytes => createHash("sha256").update(bytes).digest("hex");
const save = (name, value) => writeFileSync(join(evidence, name), JSON.stringify(value, null, 2) + "\n");
const git = (...args) => {
  const result = spawnSync("git", args, { cwd: root, maxBuffer: 128 * 1024 * 1024 });
  assert.equal(result.status, 0, result.stderr.toString());
  return result.stdout;
};
const blob = path => git("show", `${revision}:${path}`);
const json = path => JSON.parse(blob(path));
const base = "tests/integration/adapter-tools";
const service = "tests/fs/webdav/atomic-extension-independent/phase2";
const checks = [];
const environment = { ...process.env, NODE_OPTIONS: "", TMPDIR: scratch, TMP: scratch, TEMP: scratch, TSX_DISABLE_CACHE: "1" };
delete environment.NODE_TEST_CONTEXT;
function run(name, args, cwd) {
  const result = spawnSync(process.execPath, args, { cwd, env: environment, encoding: "utf8", timeout: 120_000, maxBuffer: 16 * 1024 * 1024 });
  writeFileSync(join(evidence, `${name}.stdout.log`), result.stdout ?? "");
  writeFileSync(join(evidence, `${name}.stderr.log`), result.stderr ?? "");
  const summary = { name, args, status: result.status, signal: result.signal, error: result.error?.message,
    counts: Object.fromEntries(["tests", "pass", "fail", "cancelled", "skipped", "todo"].map(key => [key, Number(new RegExp(`^# ${key} (\\d+)$`, "m").exec(result.stdout ?? "")?.[1] ?? NaN)])) };
  checks.push(summary);
  save(`${name}.result.json`, summary);
  console.log(`${name}: ${result.status}`);
  return summary;
}
let failure;

  writeFileSync(join(evidence, "audit.mjs.txt"), readFileSync(fileURLToPath(import.meta.url)));
  const seals = [];
  for (const directory of [`${base}/atomic-webdav-profile-independent`, "tests/fs/webdav/atomic-extension-independent", service]) {
    const seal = blob(`${directory}/ARTIFACTS.sha256`);

      const [expected, name] = line.split("  ");

      assert.equal(hash(blob(path)), expected, path);
      assert.equal(hash(readFileSync(join(root, path))), expected, `live historical preservation: ${path}`);
      return { path, sha256: expected };
    });
    seals.push({ path: `${directory}/ARTIFACTS.sha256`, sha256: hash(seal), rows });
  }
  save("historical-seals.json", seals);
  assert.equal(git("diff", revision, "--", `${base}/atomic-webdav-profile`, `${base}/atomic-webdav-profile-independent`, `${base}/profiles/history`).length, 0);

  }
  for (const [name, pass] of [["stock-original79", 78], ["packed-stock79", 78], ["packed-configured79", 79]]) {
    assert.deepEqual(oldAuthor.summaries[name], { tests: 79, pass, fail: 79 - pass, cancelled: 0, skipped: 0, todo: 0 });
  }
  save("historical-denominators.json", { oldAuthor, oldIndependent, wholeHistoricalRerun: false });

    const current = hash(blob(path));

    assert.equal(hash(readFileSync(join(root, path))), current, path);
    return { path, sha256: current, sameAtB22d00c: true };
  });
  save("service-provider-hashes.json", { revision, serviceRevision, providerHashes,
    fullProductChangesSinceService: git("diff", "--name-only", serviceRevision, revision, "--", "src").toString().trim().split("\n"),
    scope: "WebDAV adapter and original configured provider unchanged, not a new whole-product or deployed-provider run" });

    `${base}/atomic-webdav-profile`, `${base}/atomic-webdav-profile-independent`,
