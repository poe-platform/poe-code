async function phase(label, command, args, cwd, expected = 0) {
  assert.ok(report.phases.length < 44, "Test/tool invocation budget");
  const remaining = 15 * 60 * 1000 - (Date.now() - started);
  assert.ok(remaining > 0, "Runner wall budget");
  const stdoutPath = join(output, label + ".stdout.txt");
  const stderrPath = join(output, label + ".stderr.txt");
  const stdoutDescriptor = openSync(stdoutPath, "wx");
  const stderrDescriptor = openSync(stderrPath, "wx");
  const phaseStart = Date.now();
  const child = spawn(command, args, { cwd, env: environment, detached: true, stdio: ["ignore", stdoutDescriptor, stderrDescriptor] });
  closeSync(stdoutDescriptor);
  closeSync(stderrDescriptor);
  let timedOut = false;
  let spawnError;
  const timeout = setTimeout(() => {
    timedOut = true;
    if (child.pid && alive(child.pid)) process.kill(-child.pid, "SIGKILL");
  }, Math.min(120000, remaining));
  const result = await new Promise(resolveClose => {
    child.on("error", error => { spawnError = String(error); });
    child.on("close", (status, signal) => resolveClose({ status, signal }));
  });
  clearTimeout(timeout);
  const groupAliveOnClose = child.pid ? alive(child.pid) : false;
  if (groupAliveOnClose) {
    process.kill(-child.pid, "SIGKILL");
    for (let attempt = 0; attempt < 20 && alive(child.pid); attempt += 1) await delay(50);
  }
  const observation = { label, command, args, cwd, expected, ...result, pid: child.pid, spawnError, timedOut, groupAliveOnClose, groupSurvives: child.pid ? alive(child.pid) : false, elapsedMs: Date.now() - phaseStart, stdout: relative(output, stdoutPath), stderr: relative(output, stderrPath) };
  report.phases.push(observation);
  save("report.json", report);
  assert.equal(spawnError, undefined, label);
  assert.equal(timedOut, false, label);
  assert.equal(observation.groupSurvives, false, label);
  assert.equal(groupAliveOnClose, false, label + " leaked descendants");
  assert.equal(result.status, expected, label + " (see raw outputs)");
  return { ...observation, text: readFileSync(stdoutPath, "utf8") + readFileSync(stderrPath, "utf8") };
}

function parseDiagnostics(text) {
  const rows = [];
  for (const line of text.split("\n")) {
    const match = /^(.*)\((\d+),(\d+)\): error TS(\d+): (.*)$/.exec(line);
    if (match) rows.push({ path: match[1], line: Number(match[2]), column: Number(match[3]), code: Number(match[4]), message: match[5] });
  }
  assert.equal(rows.length, [...text.matchAll(/error TS\d+:/g)].length, "Unparsed diagnostic must not be dropped");
  return rows;
}
const diagnosticCodes = rows => rows.map(row => row.code).sort((left, right) => left - right);

function unpack(archive, destination) {
  const descriptor = openSync(archive, "r");
  const read = (length, position) => {
    const bytes = Buffer.alloc(length);
    assert.equal(readSync(descriptor, bytes, 0, length, position), length);
    return bytes;
  };
  const field = (header, start, length) => header.subarray(start, start + length).toString().replace(/\0.*$/s, "");
  const entries = [];
  let position = 0;
  let extended = {};
  try {
    while (true) {
      const header = read(512, position);
      position += 512;
      if (header.every(byte => byte === 0)) break;
      const checksum = parseInt(field(header, 148, 8).trim(), 8);
      assert.equal(header.reduce((sum, value, index) => sum + (index >= 148 && index < 156 ? 32 : value), 0), checksum);
      const size = parseInt(field(header, 124, 12).trim() || "0", 8);
      const mode = parseInt(field(header, 100, 8).trim() || "0", 8);
      const type = field(header, 156, 1);
      const bytes = read(size, position);
      position += Math.ceil(size / 512) * 512;
      if (type === "g" || type === "x") {
        const values = {};
        let offset = 0;
        while (offset < bytes.length) {
          const separator = bytes.indexOf(32, offset);
          const length = Number(bytes.subarray(offset, separator).toString());
          assert.ok(length > 0);
          const record = bytes.subarray(separator + 1, offset + length - 1).toString();
          const equals = record.indexOf("=");
          values[record.slice(0, equals)] = record.slice(equals + 1);
          offset += length;
        }
        if (type === "x") extended = values;
        continue;
      }
      const prefix = field(header, 345, 155);
      const path = extended.path ?? (prefix ? prefix + "/" : "") + field(header, 0, 100);
      extended = {};
      assert.ok(path && !path.startsWith("/") && !path.split("/").includes("..") && !path.includes("\\"));
      const target = join(destination, path);
      if (type === "5") { mkdirSync(target, { recursive: true }); continue; }
      assert.ok(type === "0" || type === "", `Nonregular archive entry ${path}: ${type}`);
      assert.ok(!existsSync(target), `Duplicate archive entry ${path}`);
      mkdirSync(dirname(target), { recursive: true });
      writeFileSync(target, bytes);
      chmodSync(target, mode & 0o777);
      assert.ok(lstatSync(target).isFile() && lstatSync(target).nlink === 1);
      entries.push(path);
    }
  } finally { closeSync(descriptor); }
  return entries;
}

function tree(revision) {
  return git("ls-tree", "-rzl", "--full-tree", revision).toString().split("\0").filter(Boolean).map(row => {
    const separator = row.indexOf("\t");
    const [mode, type, blob, size] = row.slice(0, separator).trim().split(/\s+/);
    assert.ok(["100644", "100755"].includes(mode) && type === "blob");
    return { path: row.slice(separator + 1), mode, blob, bytes: Number(size) };
  });
}

const dependencies = {};

function copyDependencies(source) {
  const sourceRoot = join(repository, "node_modules");
  function copy(local) {
    const path = join(sourceRoot, local);
    const info = lstatSync(path);
    assert.equal(info.isSymbolicLink(), false, path);
    if (info.isDirectory()) { for (const name of readdirSync(path).sort()) copy(local + "/" + name); return; }
    assert.ok(info.isFile());
    const bytes = readFileSync(path);
    const target = join(source, "node_modules", local);
    mkdirSync(dirname(target), { recursive: true });
    writeFileSync(target, bytes);
    chmodSync(target, info.mode & 0o777);
    if (dependencies[local]) assert.equal(hash(bytes), dependencies[local]);
    dependencies[local] = hash(bytes);
    assert.equal(lstatSync(target).nlink, 1);
  }
  for (const name of dependencyPackages) copy(name);
  mkdirSync(join(source, "node_modules/.bin"));
  const compiler = join(source, "node_modules/typescript/bin/tsc");
  const shim = `#!/bin/sh\nexec '${process.execPath}' '${compiler}' "$@"\n`;
  writeFileSync(join(source, "node_modules/.bin/tsc"), shim);
  chmodSync(join(source, "node_modules/.bin/tsc"), 0o755);
  return compiler;
}

    const publicResult = await run("packed-runtime", process.execPath, ["public-control.mjs", join(source, "src/index.ts")], consumer);
    snapshot.checks.public = JSON.parse(publicResult.text);
    const options = [compiler, "--noEmit", "--target", "ES2023", "--module", "NodeNext", "--moduleResolution", "NodeNext", "--strict", "--noUncheckedIndexedAccess", "--exactOptionalPropertyTypes", "--verbatimModuleSyntax", "--skipLibCheck", "false", "--types", "node", "--typeRoots", join(source, "node_modules/@types")];
    const positive = await run("packed-positive", process.execPath, [...options, "--listFiles", "aggregate.mts", "consumer.mts"], consumer);
    const typeFiles = positive.text.trim().split("\n").map(path => realpathSync(resolve(consumer, path)));
    const productDeclarations = typeFiles.filter(path => path.startsWith(join(consumer, "node_modules/virtual-bash") + "/"));
    assert.ok(productDeclarations.includes(join(consumer, "node_modules/virtual-bash/dist/index.d.ts")));
    assert.ok(productDeclarations.every(path => path.includes("/dist/") && path.endsWith(".d.ts")));
    assert.ok(typeFiles.every(path => path.startsWith(consumer + "/") || dependencyPackages.some(name => path.startsWith(join(source, "node_modules", name) + "/"))));
    snapshot.checks.packedTypeFiles = typeFiles;
    const invalid = parseDiagnostics((await run("packed-negative", process.execPath, [...options, "invalid.mts"], consumer, 2)).text);
    assert.deepEqual(diagnosticCodes(invalid), [2322, 2345, 2741]);
    assert.ok(invalid.every(row => row.path === "invalid.mts"));
    snapshot.checks.packedNegative = invalid;
    const declaration = join(consumer, "node_modules/virtual-bash/dist/index.d.ts");
    const originalDeclaration = readFileSync(declaration);
    writeFileSync(declaration, Buffer.concat([originalDeclaration, Buffer.from('\ntype IndependentBound<Value extends number> = Value;\nexport type IndependentDeclarationPoison = IndependentBound<"bad">;\n')]));
    try {
      const diagnostics = parseDiagnostics((await run("declaration-poison", process.execPath, [...options, "aggregate.mts", "consumer.mts"], consumer, 2)).text);
      assert.deepEqual(diagnosticCodes(diagnostics), [2344]);
      assert.equal(diagnostics[0].path, "node_modules/virtual-bash/dist/index.d.ts");
      snapshot.checks.declarationPoison = diagnostics;
    } finally { writeFileSync(declaration, originalDeclaration); }
    const removed = productDeclarations.map(path => ({ path, held: path + ".independent-held" }));
    for (const item of removed) renameSync(item.path, item.held);
    try {
      const diagnostics = parseDiagnostics((await run("source-fallback-negative", process.execPath, [...options, "aggregate.mts", "consumer.mts"], consumer, 2)).text);
      assert.deepEqual(diagnosticCodes(diagnostics), [7016, 7016, 7016, 7016, 7016, 7016]);
      assert.ok(diagnostics.every(row => ["aggregate.mts", "consumer.mts"].includes(row.path)));
      assert.ok(existsSync(join(source, "src/index.ts")) && existsSync(join(source, "dist/index.d.ts")));
      snapshot.checks.sourceFallback = { diagnostics, sourceAndWorkspaceDistStillPresent: true, removedDeclarationCount: removed.length };
    } finally { for (const item of removed) renameSync(item.held, item.path); }
    await run("packed-restored", process.execPath, [...options, "aggregate.mts", "consumer.mts"], consumer);
    for (const [path, expected] of Object.entries(hashes)) assert.equal(hash(readFileSync(join(source, path))), expected, path);
    for (const [path, expected] of Object.entries(dependencies)) {
      assert.equal(hash(readFileSync(join(source, "node_modules", path))), expected);
      assert.equal(hash(readFileSync(join(repository, "node_modules", path))), expected);
    }
    for (const fixture of fixtures) assert.equal(hash(readFileSync(join(consumer, fixture.destination))), hash(fixture.bytes));
    assert.equal(hash(readFileSync(declaration)), hash(originalDeclaration));
    snapshot.checks.inputsUnchangedAfterRestoration = true;
    save(label + "-results.json", snapshot);
    save("report.json", report);
    rmSync(join(root, label), { recursive: true });
  }
  report.status = "verified-bounded-configuration-only";
} catch (error) {
  report.status = "failed";
  report.error = { message: error.message, stack: error.stack };
  process.exitCode = 1;
} finally {
  save("dependency-file-hashes.json", dependencies);
  report.finalContext = context();
  save("final-context.json", report.finalContext);
  const survivingGroups = report.phases.filter(row => row.pid && alive(row.pid)).map(row => row.pid);
  for (const pid of survivingGroups) process.kill(-pid, "SIGKILL");
  rmSync(root, { recursive: true, force: true });
  report.cleanup = { survivingGroupsAtFinalCheck: survivingGroups, temporaryRootRemoved: !existsSync(root), ownedRoot: root };
  report.gitInvocations = gitInvocations;
  report.finishedAt = new Date().toISOString();
  report.elapsedMs = Date.now() - started;
  report.dependencyFilesSha256 = hash(JSON.stringify(dependencies));
  save("report.json", report);
  console.log(JSON.stringify({ status: report.status, error: report.error, phases: report.phases.length, snapshots: report.snapshots.map(row => ({ label: row.label, inclusion: { canonical: row.checks.inclusion?.canonicalIncluded, sourceTs: row.checks.inclusion?.actualSourceTs, testTs: row.checks.inclusion?.actualTestTs }, coldDiagnostics: row.checks.cold ? [row.checks.cold.original.length, row.checks.cold.fixed.length] : undefined })), cleanup: report.cleanup, elapsedMs: report.elapsedMs }, null, 2));
}
