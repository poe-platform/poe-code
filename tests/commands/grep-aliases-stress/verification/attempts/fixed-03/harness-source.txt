class ObservedWorker extends OriginalWorker {
  readonly verifierId: number;
  constructor(...args: ConstructorParameters<typeof Worker>) {
    super(...args);
    this.verifierId = ++nextWorkerIdentity;
    activeWorkers.add(this);
    workerEvents.push({ event: 'create', threadId: this.verifierId, case: currentCase, detail: String(args[0]) });
    this.once('exit', code => {
      activeWorkers.delete(this);
      workerEvents.push({ event: 'exit', threadId: this.verifierId, case: currentCase, detail: code });
    });
  }
  override postMessage(...args: Parameters<Worker['postMessage']>): void {
    workerEvents.push({ event: 'post', threadId: this.verifierId, case: currentCase });
    const send = () => { super.postMessage(...args); messageObserver?.(args[0]); };
    if (messageGate) messageGate(send); else send();
  }
  override terminate(): Promise<number> {
    workerEvents.push({ event: 'product-terminate', threadId: this.verifierId, case: currentCase });
    return super.terminate();
  }
}

function reusable(width: 4 | 8): ByteSource {
  return (async function* () {
    const shared = Buffer.alloc(width);
    const bytes = Buffer.from('keep:01\ndrop:xx\nkeep:02\n');
    try { for (let offset = 0; offset < bytes.length; offset += width) { bytes.copy(shared, 0, offset, offset + width); yield shared; } }
    finally { shared.fill(0x78); }
  })();
}

function earlySource(failure?: Error, asynchronous = false) {
  let returns = 0;
  let nextCalls = 0;
  const input: ByteSource = { [Symbol.asyncIterator]: () => ({
    next: async () => { nextCalls += 1; return { done: false, value: Buffer.from('keep:01\n') }; },
    return: () => {
      returns += 1;
      if (failure) { if (asynchronous) return Promise.reject(failure); throw failure; }
      return Promise.resolve({ done: true as const, value: undefined });
    }
  }) };
  return { input, returns: () => returns, nextCalls: () => nextCalls };
}
