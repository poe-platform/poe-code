async function fixture() {
  const sockets = new Set<Socket>();
  const traces: { method: string; path: string; body: string }[] = [];
  const server = createServer(async (request, response) => {
    const chunks: Buffer[] = [];
    for await (const chunk of request) chunks.push(Buffer.from(chunk));
    traces.push({ method: request.method!, path: request.url!, body: encode(Buffer.concat(chunks)) });
    const retry = traces.length < 3;
    const bytes = Buffer.from(retry ? "retry" : "recovered");
    response.sendDate = false;
    response.writeHead(retry ? 503 : 200, { "Retry-After": "0", "Content-Length": bytes.length, "Connection": "close" });
    response.end(bytes);
  });
  server.on("connection", (socket) => { sockets.add(socket); socket.once("close", () => sockets.delete(socket)); });
  await new Promise<void>((resolve) => server.listen(0, "127.0.0.1", resolve));
  const address = server.address();
  assert(address && typeof address !== "string");
  return { url: `http://127.0.0.1:${address.port}/retry-author`, traces, async close() { await closeResources([server], sockets); assert.equal(sockets.size, 0); } };
}

async function native(args: readonly string[], root: string, redirect = false) {
  nativeInvocations++;
  const argv = ["-q", ...args];
  const environment = { PATH: "/usr/bin:/bin", HOME: root, CURL_HOME: root, XDG_CONFIG_HOME: root, LC_ALL: "C", LANG: "C", NO_PROXY: "*", no_proxy: "*" };
  const output = redirect ? await open(join(root, "result.bin"), "w") : undefined;
  const child = spawn("/usr/bin/curl", argv, { cwd: root, env: environment, shell: false, stdio: ["ignore", output?.fd ?? "pipe", "pipe"] });
  const stdout: Buffer[] = [];
  const stderr: Buffer[] = [];
  let bytes = 0;
  let forced = false;
  const collect = (chunks: Buffer[], chunk: Buffer): void => {
    bytes += chunk.length;
    if (bytes > 1024 * 1024) { forced = true; child.kill("SIGKILL"); } else chunks.push(chunk);
  };
  child.stdout?.on("data", (chunk: Buffer) => collect(stdout, chunk));
  child.stderr!.on("data", (chunk: Buffer) => collect(stderr, chunk));
  const timer = setTimeout(() => { forced = true; child.kill("SIGKILL"); }, 7000);
  try {
    const exit = await new Promise<{ code: number | null; signal: NodeJS.Signals | null }>((resolve, reject) => {
      child.once("error", reject);
      child.once("close", (code, signal) => resolve({ code, signal }));
    });
    assert(!forced);
    return { argv, environment, stdoutFile: redirect ? "result.bin" : null, ...exit, stdout: encode(Buffer.concat(stdout)), stderr: Buffer.concat(stderr).toString() };
  } finally { clearTimeout(timer); child.kill("SIGKILL"); await output?.close(); }
}
