728:export function selectCancellationOutcome<Value>(
729:  boundary: CancellationBoundary,
730:  captured: CapturedCancellationOutcome<Value>,
731:): CancellationSelection<Value> {
732:  if (!captured || typeof captured !== "object" || captured.kind !== "return" && captured.kind !== "throw") {
733:    throw new TypeError("A captured return or throw outcome is required");
734:  }
735:  const state = boundary[boundaryState];
736:  const lineage = state.kind === "link" ? state : state.lineage;
737:  const origins = visibleOrigins(lineage);
738:  const root = origins.find(origin => origin.role === "root-caller");
739:  if (root) return throwingSelection(state, root.signal.reason, root);
740:
741:  const reported = validReport(state, captured as CapturedCancellationOutcome<unknown>);
742:  const matching = captured.kind === "throw" ? matchingVisibleOrigin(lineage, captured.reason) : undefined;
743:  const classified = reported ?? matching;
744:  const invokeMatch = classified?.role === "invoke-option";
745:  if (captured.kind === "throw" && !invokeMatch) {
746:    return throwingSelection(state, captured.reason, classified);
747:  }
748:
749:  const invokes = origins.filter(origin => origin.role === "invoke-option");
750:  if (reported?.role === "invoke-option" && !invokes.includes(reported)) invokes.push(reported);
751:  invokes.sort((left, right) => originRank(left) - originRank(right));
752:  const selectedInvoke = invokes[0];
753:  if (selectedInvoke) return throwingSelection(state, selectedInvoke.signal.reason, selectedInvoke);
754:  if (captured.kind === "throw") return throwingSelection(state, captured.reason, classified);
755:  return { outcome: { kind: "return", value: captured.value } };
