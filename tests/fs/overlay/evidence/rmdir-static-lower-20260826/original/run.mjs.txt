import assert from 'node:assert/strict';
import { readFileSync, writeFileSync } from 'node:fs';
import { execFileSync, spawnSync } from 'node:child_process';
import { createHash } from 'node:crypto';
import { dirname, join } from 'node:path';
import { fileURLToPath } from 'node:url';
const root = dirname(fileURLToPath(import.meta.url));
const repo = '/Users/kjopek/Workspace/safe-bash';
const archive = join(root, 'archive');
const pin = '3a9177a7b8ddc1c20142720ffa96ea6f87f65e86';
const hash = data => createHash('sha256').update(data).digest('hex');
const git = (...args) => execFileSync('git', args, { cwd: repo });
const save = (name, value) => writeFileSync(join(root, name), JSON.stringify(value, null, 2) + '\n');
const paths = git('ls-tree', '-r', '--name-only', pin, '--', 'src/contracts', 'src/fs/overlay', 'src/fs/memory', 'package.json', 'package-lock.json').toString().trim().split('\n');
const manifest = () => paths.map(path => ({ path, sha256: hash(readFileSync(join(archive, path))), blob: git('rev-parse', `${pin}:${path}`).toString().trim() }));
const before = manifest();
for (const entry of before) assert.equal(entry.sha256, hash(git('show', `${pin}:${entry.path}`)));
save('manifest-before.json', before);
const args = ['--import', 'tsx', '../probe.mjs'];
const start = new Date().toISOString();
const result = spawnSync(process.execPath, args, { cwd: archive, encoding: 'utf8', timeout: 15000, env: { ...process.env, TMPDIR: root, TSX_DISABLE_CACHE: '1' } });
writeFileSync(join(root, 'probe.stdout'), result.stdout ?? '');
writeFileSync(join(root, 'probe.stderr'), result.stderr ?? '');
save('run.json', { pin, node: process.version, tsx: JSON.parse(readFileSync(join(archive, 'node_modules/tsx/package.json'))).version, start, end: new Date().toISOString(), argv: [process.execPath, ...args], cwd: archive, status: result.status, signal: result.signal, error: result.error?.message, sourceTarSha256: hash(readFileSync(join(root, 'source.tar'))), probeSha256: hash(readFileSync(join(root, 'probe.mjs'))) });
const after = manifest();
assert.deepEqual(after, before);
save('manifest-after.json', after);
console.log(JSON.stringify({ status: result.status, sourceFilesUnchanged: before.length }));
if (result.status !== 0) process.exitCode = 1;
