import * as shell from "shell";
import * as command from "command";
let assertions = 0;
function check(condition, label) {
  if (!condition) throw new Error(label);
  assertions += 1;
}
check(!Object.hasOwn(command.env, "OPTIND") && !Object.hasOwn(command.env, "OPTERR"), "outer locals not exported");
const result = await shell.exec(`set -- -ab 'λ value' tail
while getopts 'ab:' option; do
  status=$?
  printf 'G1|step|%s|%s|%s|%s|%s|%s|%s|%s\n' "$status" "$option" "\${OPTARG-unset}" "$OPTIND" "$#" "$1" "$2" "$3"
done
getopts 'ab:' option
status=$?
printf 'G1|end|%s|%s|%s|%s|%s|%s|%s|%s\n' "$status" "$option" "\${OPTARG-unset}" "$OPTIND" "$#" "$1" "$2" "$3"`);
check(result.exitCode === 0, "script status");
check(result.stderr === "", "no diagnostics");
check(result.stdout === "G1|step|0|a|unset|1|3|-ab|λ value|tail\nG1|step|0|b|λ value|3|3|-ab|λ value|tail\nG1|end|1|?|unset|3|3|-ab|λ value|tail\n", "cluster argument state and termination");
return { marker: "G1_GUEST_ASSERTIONS_COMPLETE", assertions, observations: [result] };
